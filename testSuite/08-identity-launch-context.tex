\section{Identity Launch Context}


\begin{javascript}
function runtimeLaunchValue(name) {
  var activity = document.querySelector("main.activity");
  if (!activity)
    return "missing-activity";

  var value = activity.getAttribute(name);
  return value===null ? "missing" : value;
}

function runtimeIdentityLabel() {
  var candidates = [
    document.querySelector("[data-user-id]"),
    document.querySelector(".user-name"),
    document.querySelector("#user-name")
  ];

  for (var i=0;i<candidates.length;i++) {
    var element = candidates[i];
    if (!element)
      continue;

    var value =
      element.getAttribute("data-user-id") ||
      element.textContent;

    if (value && value.trim())
      return value.trim();
  }

  return "identity-not-visible";
}
\end{javascript}
