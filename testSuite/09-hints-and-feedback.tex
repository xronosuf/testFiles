\section{Optional Interactive}

\begin{javascript}
function installRuntimeOptionalInteractive() {
  var host = document.getElementById(
    "runtime-optional-interactive"
  );

  if (
    !host ||
    host.getAttribute("data-installed") === "true"
  ) {
    return;
  }

  host.setAttribute(
    "data-installed",
    "true"
  );

  var button = document.createElement("button");
  button.type = "button";
  button.className = "btn btn-primary";
  button.textContent = "Increment optional counter";

  var output = document.createElement("span");
  output.style.marginLeft = "1em";
  output.textContent = "0";

  var count = 0;

  button.addEventListener(
    "click",
    function() {
      count += 1;
      output.textContent = String(count);
    }
  );

  host.textContent = "";
  host.appendChild(button);
  host.appendChild(output);
}

setTimeout(
  installRuntimeOptionalInteractive,
  0
);
\end{javascript}
